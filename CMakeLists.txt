cmake_minimum_required(VERSION 3.20)
project(lock_free_queues LANGUAGES CXX)

add_library(lfq INTERFACE)
target_include_directories(lfq INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(lfq INTERFACE cxx_std_20)

include(CTest)
option(LFQ_ENABLE_TSAN "Enable ThreadSanitizer in tests" OFF)
option(LFQ_ENABLE_ASAN_UBSAN "Enable AddressSanitizer and UndefinedBehaviorSanitizer in tests" OFF)
if(LFQ_ENABLE_TSAN AND LFQ_ENABLE_ASAN_UBSAN)
    message(FATAL_ERROR "Use separate build directories for TSan and ASan/UBSan")
endif()

if(BUILD_TESTING)
    find_package(Threads REQUIRED)
    if(LFQ_ENABLE_TSAN OR LFQ_ENABLE_ASAN_UBSAN)
        if(WIN32 OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
            message(FATAL_ERROR "These sanitizer presets require a supported non-Windows GCC/Clang toolchain")
        endif()
        if(LFQ_ENABLE_TSAN)
            set(LFQ_SANITIZER thread)
        else()
            set(LFQ_SANITIZER address,undefined)
        endif()
    endif()
    foreach(suite IN ITEMS spsc_ring concurrent_ring)
        add_executable(${suite}_tests tests/${suite}_tests.cpp)
        target_link_libraries(${suite}_tests PRIVATE lfq Threads::Threads)
        set_target_properties(${suite}_tests PROPERTIES CXX_EXTENSIONS OFF)
        if(MSVC)
            target_compile_options(${suite}_tests PRIVATE /W4 /permissive-)
        else()
            target_compile_options(${suite}_tests PRIVATE -Wall -Wextra -Wpedantic)
        endif()
        if(LFQ_ENABLE_TSAN OR LFQ_ENABLE_ASAN_UBSAN)
            target_compile_options(${suite}_tests PRIVATE -fsanitize=${LFQ_SANITIZER} -fno-omit-frame-pointer -g)
            target_link_options(${suite}_tests PRIVATE -fsanitize=${LFQ_SANITIZER})
        endif()
        add_test(NAME ${suite} COMMAND ${suite}_tests)
        set_tests_properties(${suite} PROPERTIES TIMEOUT 60)
    endforeach()
endif()
